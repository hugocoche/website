tab_contingence <- function(x,titre=NULL,...){
  x<- addmargins(as.matrix(x), FUN = list(Total = sum), quiet = TRUE)

  # Renommer la dernière ligne et la dernière colonne
  rownames(x)[nrow(x)] <- "Total"
  colnames(x)[ncol(x)] <- "Total"
  x %>% 
    kable() %>% 
    kable_styling(
      bootstrap_options = c("striped", "hover", "condensed"),
      full_width = F, 
      position = "center") 
}

tableau_contingence_latex <- function(x,titre=NULL,position = 'center',...){
  x %>% 
    kable(caption = titre,
        align = "c",
        ...) |> 
  kable_styling(latex_options =  c("HOLD_position","scale_down","striped","hover"),
                full_width = FALSE,
                position = position) |> 
    row_spec(0,
           background = "cornflowerblue",
           color = "white",
           italic = FALSE,
           bold = TRUE) %>% 
    column_spec(1,
                background = "blanchedalmond",
                color = "black",
                italic = TRUE)

}


graph_pointhtml <- function(tab, titre_legende = "Légende", titre = NULL, sous_titre = NULL, xlab = "x", ylab = "y", abs, ord, quali_var = NULL, cond = FALSE, ...) {
  # color ou fill, pas les deux

    aes_string(x = abs, y = ord)
  }
  
  tab %>% 
    ggplot(aes_string(x = abs, y = ord, color = "quali_var")) +
    geom_point(aes_string(x = abs, y = ord), show.legend = cond) +
    labs(x = xlab, y = ylab, title = titre, subtitle = sous_titre,
         color = titre_legende) +  # Modifier le titre de la légende
    theme(plot.title = element_text(color = "black", hjust = 0.5, face = "bold.italic"),
          plot.subtitle = element_text(color = "red", hjust = 0.5, face = "bold.italic"),
          panel.background = element_rect(fill = "lightgrey"),

matrice_cor <- function(tab,titre = NULL){
  corrplot(cor(tab),type = "upper",addCoef.col = "black",
           tl.cex = 0.75,number.cex = 0.75,tl.col = "black",
           method = "square",
           title = titre,
           diag = FALSE)
  
}

top_chart <- function(x) {
  case_when(
    x == 0 ~ 'pas classé',
    between(x, 1, 10) ~ 'Top 10',
    between(x, 11, 50) ~ 'Top 50',
    between(x, 51, 200) ~ 'Top 200',
    between(x, 101, 1000000) ~ '> 200',
    TRUE ~ '?'
  )
}

top_chart <- function(x) {
  case_when(
    x == 0 ~ 'pas classé',
    between(x, 1, 5) ~ 'Top 5',
    between(x, 5.1, 20) ~ 'Top 20',
    between(x, 20.0001, 50) ~ 'Top 50',
    between(x, 50.000001, 80000) ~ '>50',
    TRUE ~ '?'
  )
}
